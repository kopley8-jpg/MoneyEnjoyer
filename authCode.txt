import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet } from 'react-native';
import {
  FirebaseAuthTypes,
  getAuth,
  onAuthStateChanged,
} from '@react-native-firebase/auth';
import {
  GoogleOneTapSignIn,
} from 'react-native-nitro-google-signin';
import { startSignInFlow } from '@/shared/api/firebase/googleSignIn';
import { authInstance } from '@/shared/api/firebase/firebase';

GoogleOneTapSignIn.configure({ webClientId: 'autoDetect' });


const App = () => {

  const [user, setUser] = useState<FirebaseAuthTypes.User | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(getAuth(), (user) => {
      console.log(user)
      setUser(user)
      if(loading) setLoading(false)
    })
  return unsubscribe
  }, [])

  const handleTextPress = async () => {
    if(!user){
      startSignInFlow()
    }else{
      console.log("signing out")
      await authInstance.signOut()
      await GoogleOneTapSignIn.signOut()
      console.log(user)
    }
  }

  return(
    <View style={styles.container}>
      <Text onPress={handleTextPress}>{user?"welcome"+user.email:"login"}</Text>
    </View>
  )
}

export default App;

const styles = StyleSheet.create({
  container:{
    width:"100%",
    height:"100%",
    backgroundColor:"white",
    display:"flex",
    alignItems:"center",
    justifyContent:"center"
  }
})
